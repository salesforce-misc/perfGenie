dashboard.fetchData() Method

/**
 * Fetch data from a REST endpoint for a panel target
 * 
 * @param {string} endpoint - REST endpoint URL (with QEURY placeholder replaced by URL-encoded query)
 * @param {string} query - Processed query string (with $variables replaced)
 * @param {number} [startTimestamp=null] - Optional start timestamp for the data range (in milliseconds)
 * @param {number} [endTimestamp=null] - Optional end timestamp for the data range (in milliseconds)
 * @returns {Promise<Array|Object>} - Promise resolving to data in one of the supported formats:
 *   - Array of series objects (recommended)
 *   - Single series object with datapoints
 *   - Single series object with times/values arrays
 * 
 * @description
 * This method is called automatically by the component for each target in a panel.
 * You can override it to provide custom data fetching logic or mock data.
 * 
 * The method receives:
 * - endpoint: Full URL with query encoded (e.g., "/api/v1/query?query=SELECT%20*%20FROM...")
 * - query: The processed SQL/query string with $variables replaced
 * - startTimestamp: Start time in milliseconds (if available)
 * - endTimestamp: End time in milliseconds (if available)
 * 
 * For "Compare with previous" feature, the timestamps represent the period being queried
 * (which may be shifted for previous period comparisons).
 * 
 * Return formats:
 * 1. Array of series (recommended - matches safepoint.json format):
 *    [
 *      {
 *        scope: "core.aws.aws-prod0-uswest2.core1",
 *        metric: "cpu_usage",
 *        tags: { cell: "usa12", role: "app", ... },
 *        displayName: "Optional - auto-generated if not provided",
 *        datapoints: { "timestamp": value, ... }
 *      },
 *      ...
 *    ]
 * 
 * 2. Single series with array datapoints:
 *    {
 *      displayName: "Series Name",
 *      datapoints: [[value, timestamp], [value, timestamp], ...]
 *    }
 * 
 * 3. Single series with object datapoints:
 *    {
 *      displayName: "Series Name",
 *      datapoints: { "timestamp": value, ... }
 *    }
 * 
 * 4. Single series with times/values arrays:
 *    {
 *      displayName: "Series Name",
 *      times: [timestamp1, timestamp2, ...],
 *      values: [value1, value2, ...]
 *    }
 * 
 * Example - Override with mock data:
 */
const dashboard = new GenieDashboard('dashboard-container');

dashboard.fetchData = async function(endpoint, query, startTimestamp = null, endTimestamp = null) {
    // Use provided timestamps if available, otherwise use defaults
    const queryStart = startTimestamp || (Date.now() - 3600000);
    const queryEnd = endTimestamp || Date.now();
    
    // Generate mock time series data for the time range
    const datapoints = {};
    const dataPointCount = 20;
    const timeRange = queryEnd - queryStart;
    const interval = Math.max(60000, Math.floor(timeRange / dataPointCount));
    
    for (let i = 0; i < dataPointCount; i++) {
        const timestamp = queryStart + (i * interval);
        datapoints[timestamp] = Math.random() * 100 + 50;
    }
    
    // Return array format (recommended)
    return [{
        scope: 'core.aws.aws-prod0-uswest2.core1',
        metric: 'cpu_usage',
        tags: {
            cell: 'usa12',
            role: 'app',
            service: 'coreapp'
        },
        // displayName will be auto-generated from scope:metric:{tags}
        datapoints: datapoints  // Object format: { "timestamp": value }
    }];
};

await dashboard.render(dashboardJson, inputJson);

Example with Mock Data (No API Calls)

// Override fetchData method to return mock data
const dashboard = new GenieDashboard('dashboard-container');

// Track fetch calls to handle multiple targets and previous period detection
let fetchCallCount = 0;
let timeseriesFetchCount = 0; // Track only timeseries panel calls (exclude stat panel)
let originalStartValue = null;
let originalEndValue = null;
let currentPeriodRange = null;

dashboard.fetchData = async function(endpoint, query, startTimestamp = null, endTimestamp = null) {
    fetchCallCount++;
    
    // Check if this is for a stat panel
    if (query && query.includes('COUNT')) {
        // For stat panels, return a single value
        let isPreviousPeriod = false;
        // Detect previous period (simplified - check timestamp difference)
        if (originalStartValue && startTimestamp && startTimestamp < originalStartValue - 3600000) {
            isPreviousPeriod = true;
        }
        return {
            displayName: 'Total Requests',
            datapoints: {
                [Date.now()]: Math.floor(Math.random() * 10000) + (isPreviousPeriod ? 4000 : 6000)
            }
        };
    }
    
    // Track timeseries panel calls
    timeseriesFetchCount++;
    // Determine which target based on call order
    // Odd calls (1, 3, 5...) = target A, Even calls (2, 4, 6...) = target B
    const isTargetA = (timeseriesFetchCount % 2 === 1);
    const targetMetric = isTargetA ? 'cpu_usage' : 'memory_usage';
    
    // Detect previous period
    let isPreviousPeriod = false;
    if (originalStartValue && startTimestamp && startTimestamp < originalStartValue - 3600000) {
        isPreviousPeriod = true;
    }
    
    // Use provided timestamps if available, otherwise use defaults
    let queryStart = startTimestamp || (Date.now() - 3600000);
    let queryEnd = endTimestamp || Date.now();
    
    // Store original period on first call
    if (!originalStartValue) {
        originalStartValue = queryStart;
        originalEndValue = queryEnd;
        currentPeriodRange = queryEnd - queryStart;
    }
    
    // Generate multiple series for the target metric
    const response = [];
    const scopes = ['core.aws.aws-prod0-uswest2.core1', 'core.aws.aws-prod0-uswest2.core2'];
    const cells = ['usa12', 'usa13'];
    const roles = ['app', 'api'];
    const seriesCount = 1; // 1 series per target
    
    for (let i = 0; i < seriesCount; i++) {
        const scope = scopes[i % scopes.length];
        const cell = cells[i % cells.length];
        const role = roles[i % roles.length];
        
        // Generate datapoints for this series
        const datapoints = {};
        const dataPointCount = 20;
        const timeRange = queryEnd - queryStart;
        const interval = Math.max(60000, Math.floor(timeRange / dataPointCount));
        
        // Different base values for different metrics
        let baseValue = 50;
        if (targetMetric === 'cpu_usage') {
            baseValue = 60 + (i * 15); // CPU usage: 60-75%
        } else if (targetMetric === 'memory_usage') {
            baseValue = 70 + (i * 20); // Memory usage: 70-90%
        }
        
        for (let j = 0; j < dataPointCount; j++) {
            const timestamp = queryStart + (j * interval);
            const variation = (Math.random() * 20) - 10; // ±10 variation
            const adjustedValue = isPreviousPeriod 
                ? (baseValue + variation) * (0.9 + Math.random() * 0.2) 
                : baseValue + variation;
            datapoints[timestamp] = Math.max(0, adjustedValue);
        }
        
        response.push({
            scope: scope,
            metric: targetMetric,
            tags: {
                cell: cell,
                role: role,
                service: 'coreapp',
                pod: cell + '-pod-' + (i % 3),
                k8s_pod_name: cell + '-casam-app-blue-' + (i % 2),
                environment: 'prod'
            },
            // displayName will be auto-generated from scope:metric:{tags}
            datapoints: datapoints  // Object format: { "timestamp": value }
        });
    }
    
    return response; // Return array format
};

await dashboard.render(dashboardJson, inputJson);

FieldConfig Overrides for Series Names

// Use fieldConfig.overrides to customize series display names and colors
// This is useful when API returns series with long names or you want user-friendly labels

fieldConfig: {
    defaults: {
        unit: 'ms',
        custom: {
            drawStyle: 'line',
            lineWidth: 0.75
        }
    },
    overrides: [
        {
            // Match series by regex pattern (case-insensitive)
            matcher: { 
                id: 'byRegexp',           // Matcher type: regex pattern
                options: '/cpu_usage|CPU Usage/'  // Regex pattern (remove slashes for matching)
            },
            properties: [
                {
                    id: 'displayName',    // Property to override
                    value: 'CPU Usage'     // Custom display name
                },
                {
                    id: 'color',          // Optional: override color
                    value: { 
                        fixedColor: 'red', 
                        mode: 'fixed' 
                    }
                }
            ]
        },
        {
            matcher: { id: 'byRegexp', options: '/memory_usage|Memory Usage/' },
            properties: [
                {
                    id: 'displayName',
                    value: 'Memory Usage'
                },
                {
                    id: 'color',
                    value: { fixedColor: 'blue', mode: 'fixed' }
                }
            ]
        }
    ]
}

// How it works:
// 1. The matcher uses 'byRegexp' to match series names against a regex pattern
// 2. The pattern can match against:
//    - Original series name (from API response)
//    - Metric name (e.g., "cpu_usage")
//    - Display name (e.g., "CPU Usage")
//    - Built display name (e.g., "core.aws...:cpu_usage:{tags}")
// 3. When a match is found, the override properties are applied
// 4. displayName override replaces the series name in legends, tooltips, and statistics
// 5. color override can set a fixed color for the series

// Example: Series with name "core.aws.prod:cpu_usage:{cell=usa12}" will match
// pattern "/cpu_usage/" and display as "CPU Usage" in the chart

// Multiple overrides are checked in order - first match wins

Compare with Previous Period

// Add "previous" key to inputJson to enable period comparison dropdown
const inputJson = {
    argus: '/api/v1/query?query=QEURY',
    '$start': Date.now() - 3600000,  // Current period start
    '$end': Date.now(),               // Current period end
    'stats': ['avg', 'max', 'min', 'p90', 'p95', 'p99'],
    'previous': 'none,-1d,-7d,-14d,-21d,-28d'  // Comma-separated list: "none" + durations with "-" prefix
    // Format: "none" + comma-separated durations like '-1d', '-7d', '-14d', '-21d', '-28d', '-1w', '-1m'
};

// When "previous" is specified:
// 1. A dropdown appears next to the tabs with options: None, -1d, -7d, -14d, -21d, -28d
// 2. When a duration is selected, all queries are re-run with $start and $end reduced by the duration
// 3. Previous period series are suffixed with "{duration}" (e.g., "Series Name(-7d)")
//    - This format groups related series together in tooltips and legends
// 4. Stats table shows current value with % change in brackets:
//    - Green for increases: "100.00 (+5.2%)"
//    - Red for decreases: "95.50 (-4.8%)"

// Example with multiple duration options:
{
    "argus": "/v1/geniequery/?query=QEURY",
    "$start": 1762106820000,
    "$end": 1762711620000,
    "stats": ["avg", "max", "min", "p90", "p95", "p99"],
    "previous": "none,-1d,-7d,-14d,-21d,-28d"  // Dropdown with None, -1d, -7d, -14d, -21d, -28d
}

// When a duration is selected:
// - Chart shows both current and previous series (previous suffixed with "(-{duration})")
//   Example: "CPU Usage" and "CPU Usage(-7d)" appear next to each other in tooltips/legends
// - Statistics table shows: "Current Value (+X.X%)" or "Current Value (-X.X%)"
// - Percentage change calculated as: ((current - previous) / previous) * 100
