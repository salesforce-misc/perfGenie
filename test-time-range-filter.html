💡 Example Usage in Code

// Basic usage
showTimeRangeFilter(startTimestamp, endTimestamp, 
    (result) => {
        console.log('Selected range:', result);
        // result.start and result.end are epoch timestamps
    },
    () => {
        console.log('Cancelled');
    }
);

// With label prefix
showTimeRangeFilter(startTimestamp, endTimestamp, 
    (result) => {
        console.log('Selected range:', result);
    },
    () => {
        console.log('Cancelled');
    },
    'Event' // This will show "Event Start Time:" and "Event End Time:"
);

// Example with current time
const now = Date.now();
const yesterday = now - (24 * 60 * 60 * 1000);
showTimeRangeFilter(yesterday, now, onApply, onCancel);
